#include<stdio.h>
#include<signal.h>
#include<sys/wait.h>
#include<unistd.h>
int main() {
pid_t pid;
printf("Parent Process ID = %d\n",getpid());
//1.Genarate SIGINT
pid = fork();
if(pid==0){
printf("\n Generating SIGINT...\n");
raise(SIGINT);
printf("\n Generated SIGINT...\n");
}
//2.Terminate
pid=fork();
if(pid==0){
printf("\n Generating SIGTERM...\n");
raise(SIGTERM);
printf("\n Generated SIGTERM...\n");
}
//3.killing
pid=fork();
if(pid==0){
printf("\n Generating SIGKILL...\n");
raise(SIGKILL);
printf("\nG enerated SIGKILL...\n");
}
//4.Stoping
pid=fork();
if(pid==0){
printf("\n Generating SIGSTOP...\n");
raise(SIGSTOP);
printf("\n Generated SIGSTOP...\n");
}
return 0;
}
