#include<stdio.h>
#include<fcntl.h>
#include<unistd.h>

int main(){
int s,d,n;
char b[100];

s=open("private.txt",O_RDONLY);
if(s<0){
printf("Source file error");
return 1;
}

d=open("public.txt",O_WRONLY|O_CREAT|O_TRUNC,0644);
if(d<0){
printf("Destination file error");
close(s);
return 1;
}

while((n=read(s,b,100))>0){
if(write(d,b,n)!=n){
printf("Write error");
return 1;
}
}

close(s);
close(d);

printf("File copied successfully\n");
return 0;
}
